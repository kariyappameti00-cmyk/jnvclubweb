Sample code

<html>
    <head>
        <title>My Web Page</title>
    </head>
    <body>
        <header>
            <h1>My Web Page</h1>
        </header>
        <main>
            <p>This is a simple web page.</p>
        </main>
        <footer>
            <p>© 2023 My Web Page. All rights reserved.</p>
        </footer>
    </body>
</html>
